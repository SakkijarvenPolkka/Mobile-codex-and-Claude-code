#[[
SPDX-License-Identifier: GPL-2.0-or-later

Audacity Android port -- the C++ bridge (native/bridge), see MODULES.md.

Targets
  aubridge-headers   INTERFACE: include dirs (include/, core/, bridge root,
                     third_party/) + audacity-core.  Everything below links it.
  aubridge-core      STATIC (PIC): core/*.cpp -- the spine.
  aubridge-<module>  STATIC (PIC), one per module directory (edit effects io
                     audio display); sources from <module>/sources.cmake:
                        AUBRIDGE_<MODULE>_SOURCES            all platforms
                        AUBRIDGE_<MODULE>_ANDROID_SOURCES    Android only
                        AUBRIDGE_<MODULE>_LIBRARIES          extra link libs
                        AUBRIDGE_<MODULE>_ANDROID_LIBRARIES  extra link libs (Android)
                     Relative source paths are resolved against the module
                     directory (prefer ${CMAKE_CURRENT_LIST_DIR}/x.cpp).
  audacity-bridge    SHARED (libaudacity-bridge.so): whole-archive of the core
                     and every module + audacity-core, linked with
                     --no-undefined.  JNI and the host tests link this.

The static libraries link aubridge-headers rather than aubridge-core: the
whole-archive link of audacity-bridge must be the only place where
aubridge-core appears on a link line (CMake rejects one library linked both
with and without the WHOLE_ARCHIVE feature).
]]

set(AUBRIDGE_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
set(AUBRIDGE_MODULES edit effects io audio display)

# ---------------------------------------------------------------------------
# Build information header (app.info)
# ---------------------------------------------------------------------------
get_property(_aubridge_libs GLOBAL PROPERTY AUDACITY_ALL_LIBRARIES)
get_property(_aubridge_mods GLOBAL PROPERTY AUDACITY_ALL_MODULES)
set(_aubridge_lib_list "")
foreach(_l IN LISTS _aubridge_libs _aubridge_mods)
   get_target_property(_type ${_l} TYPE)
   if(NOT _type STREQUAL "INTERFACE_LIBRARY")
      string(APPEND _aubridge_lib_list "   \"${_l}\", \\\n")
   endif()
endforeach()
if(ANDROID)
   set(AUBRIDGE_ABI "${ANDROID_ABI}")
else()
   set(AUBRIDGE_ABI "${CMAKE_SYSTEM_NAME}-${CMAKE_SYSTEM_PROCESSOR}")
endif()
set(AUBRIDGE_ENGINE_VERSION "1.0.0")
set(_aubridge_info_header "${CMAKE_CURRENT_BINARY_DIR}/generated/aubridge_build_info.h")
file(CONFIGURE OUTPUT "${_aubridge_info_header}" CONTENT
"// Generated by native/bridge/CMakeLists.txt -- do not edit
#pragma once
#define AUBRIDGE_ENGINE_VERSION \"@AUBRIDGE_ENGINE_VERSION@\"
#define AUBRIDGE_ABI \"@AUBRIDGE_ABI@\"
#define AUBRIDGE_LIBRARY_NAMES \\
@_aubridge_lib_list@   nullptr
"
   @ONLY)

# ---------------------------------------------------------------------------
# Usage requirements shared by the core and the modules
# ---------------------------------------------------------------------------
add_library(aubridge-headers INTERFACE)
target_include_directories(aubridge-headers INTERFACE
   "${AUBRIDGE_DIR}/include"
   "${AUBRIDGE_DIR}/core"
   "${AUBRIDGE_DIR}"
   "${AUBRIDGE_DIR}/third_party"
   "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_link_libraries(aubridge-headers INTERFACE audacity-core)
target_compile_definitions(aubridge-headers INTERFACE
   # nlohmann::json: keep implicit conversions explicit
   JSON_USE_IMPLICIT_CONVERSIONS=0)
if(ANDROID)
   target_link_libraries(aubridge-headers INTERFACE log)
endif()

function(_aubridge_resolve_sources out module_dir)
   set(result)
   foreach(src IN LISTS ARGN)
      if(IS_ABSOLUTE "${src}")
         list(APPEND result "${src}")
      elseif(EXISTS "${module_dir}/${src}")
         list(APPEND result "${module_dir}/${src}")
      elseif(EXISTS "${module_dir}/android/${src}")
         # Android lists written relative to <module>/android/
         list(APPEND result "${module_dir}/android/${src}")
      else()
         list(APPEND result "${module_dir}/${src}")
      endif()
   endforeach()
   set(${out} "${result}" PARENT_SCOPE)
endfunction()

# ---------------------------------------------------------------------------
# Spine
# ---------------------------------------------------------------------------
file(GLOB AUBRIDGE_CORE_SOURCES CONFIGURE_DEPENDS
   "${AUBRIDGE_DIR}/core/*.cpp"
   "${AUBRIDGE_DIR}/core/*.h")
add_library(aubridge-core STATIC
   ${AUBRIDGE_CORE_SOURCES}
   "${AUBRIDGE_DIR}/include/aubridge/Bridge.h")
target_link_libraries(aubridge-core PUBLIC aubridge-headers)
set_target_properties(aubridge-core PROPERTIES POSITION_INDEPENDENT_CODE ON)

# ---------------------------------------------------------------------------
# Feature modules
# ---------------------------------------------------------------------------
set(_aubridge_archives aubridge-core)
foreach(_m IN LISTS AUBRIDGE_MODULES)
   string(TOUPPER "${_m}" _M)
   set(AUBRIDGE_${_M}_SOURCES)
   set(AUBRIDGE_${_M}_ANDROID_SOURCES)
   set(AUBRIDGE_${_M}_LIBRARIES)
   set(AUBRIDGE_${_M}_ANDROID_LIBRARIES)
   include("${AUBRIDGE_DIR}/${_m}/sources.cmake")
   set(_srcs ${AUBRIDGE_${_M}_SOURCES})
   if(ANDROID)
      list(APPEND _srcs ${AUBRIDGE_${_M}_ANDROID_SOURCES})
   endif()
   _aubridge_resolve_sources(_srcs "${AUBRIDGE_DIR}/${_m}" ${_srcs})
   add_library(aubridge-${_m} STATIC ${_srcs})
   target_link_libraries(aubridge-${_m} PUBLIC aubridge-headers ${AUBRIDGE_${_M}_LIBRARIES})
   if(ANDROID)
      target_link_libraries(aubridge-${_m} PUBLIC ${AUBRIDGE_${_M}_ANDROID_LIBRARIES})
      # NDK media codecs for the Android import/export plug-ins (io/android)
      if(_m STREQUAL "io" AND AUBRIDGE_IO_ANDROID_SOURCES)
         target_link_libraries(aubridge-io PUBLIC mediandk)
      endif()
   endif()
   set_target_properties(aubridge-${_m} PROPERTIES POSITION_INDEPENDENT_CODE ON)
   list(APPEND _aubridge_archives aubridge-${_m})
endforeach()

# ---------------------------------------------------------------------------
# libaudacity-bridge.so
# ---------------------------------------------------------------------------
set(_aubridge_stub "${CMAKE_CURRENT_BINARY_DIR}/generated/audacity_bridge_so.cpp")
file(CONFIGURE OUTPUT "${_aubridge_stub}" CONTENT
"// Generated: libaudacity-bridge.so consists of whole archives only
// SPDX-License-Identifier: GPL-2.0-or-later
")
add_library(audacity-bridge SHARED "${_aubridge_stub}")
list(JOIN _aubridge_archives "," _aubridge_archive_list)
target_link_libraries(audacity-bridge
   PRIVATE "$<LINK_LIBRARY:WHOLE_ARCHIVE,${_aubridge_archive_list}>" audacity-core)
target_include_directories(audacity-bridge PUBLIC "${AUBRIDGE_DIR}/include")
if(NOT ANDROID AND CMAKE_SYSTEM_NAME STREQUAL "Linux")
   # Also enforced globally by the superbuild (CMAKE_SHARED_LINKER_FLAGS)
   target_link_options(audacity-bridge PRIVATE "LINKER:--no-undefined")
endif()
set_target_properties(audacity-bridge PROPERTIES
   LIBRARY_OUTPUT_DIRECTORY "${AUDACITY_LIB_OUTPUT_DIR}"
   RUNTIME_OUTPUT_DIRECTORY "${AUDACITY_LIB_OUTPUT_DIR}")
if(NOT ANDROID)
   set_target_properties(audacity-bridge PROPERTIES
      BUILD_RPATH "$ORIGIN" INSTALL_RPATH "$ORIGIN")
endif()
